3. Google Таблица как База Данных

Хранение всех цифр в Google Sheets для сайта на Vercel

Статус: Данные хранятся локально в браузере.
📖 Показать код скрипта и инструкцию подключения Google Таблиц

1. Откройте новую Google Таблицу → Расширения → Apps Script.

2. Вставьте код ниже и нажмите Начать развертывание → Веб-приложение (Доступ: Все / Anyone). Вставьте ссылку выше.

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('DB_JSON');
  if (!sheet) return ContentService.createTextOutput('{}').setMimeType(ContentService.MimeType.JSON);
  return ContentService.createTextOutput(sheet.getRange('A2').getValue() || '{}').setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var raw = e.postData.contents;
  var data = JSON.parse(raw);

  // 1. Технический лист хранения состояния для сайта
  var db = ss.getSheetByName('DB_JSON') || ss.insertSheet('DB_JSON');
  db.clear();
  db.getRange('A1').setValue('JSON State');
  db.getRange('A2').setValue(raw);
  db.getRange('B1').setValue('Updated');
  db.getRange('B2').setValue(new Date());

  // 2. Наглядная таблица рейтинга сотрудников
  var lb = ss.getSheetByName('Рейтинг') || ss.insertSheet('Рейтинг');
  lb.clear();
  var headers = ['Сотрудник', 'Часы', 'Сим', 'Абоны', 'Товарная выручка', 'Аксессуары', 'Фин услуги', 'Мега семья', 'Адаптер'];
  lb.appendRow(headers);
  if (data.period && data.period.staff) {
    data.period.staff.forEach(function(s) {
      var f = s.facts || {};
      lb.appendRow([s.name, s.hours || 0, f.sim||0, f.abony||0, f.tovarka||0, f.acc||0, f.fin||0, f.mega||0, f.adapter||0]);
    });
  }
  return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
}
