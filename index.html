🚀 Как подключить Google Таблицу за 2 минуты

Шаг 1. Создайте новую пустую Google Таблицу (sheets.new). В верхнем меню нажмите Расширения → Apps Script.
Шаг 2. Удалите весь текст в редакторе и вставьте готовый код ниже. Он сам создаст наглядные листы («Рейтинг_Конкурса», «План_Конкурса», «Дневной_План», «База_JSON»), чтобы вы могли смотреть и править цифры прямо из Google Таблицы!
Шаг 3. Нажмите синюю кнопку «Начать развертывание» → «Новое развертывание». Выберите тип «Веб-приложение»:
  • Запуск от имени: Меня (Me)
  • У кого есть доступ: Все (Anyone) — обязательно!
Скопируйте полученный URL (заканчивается на /exec) и вставьте в поле выше.
Готовый код для Apps Script: 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName('База_JSON');
  if (!dbSheet) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'empty' }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var rawJson = dbSheet.getRange('A2').getValue();
  return ContentService.createTextOutput(rawJson || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var body = e.postData.contents;
  var data = JSON.parse(body);

  // 1. Сохраняем полный слепок состояния для сайта на Vercel
  var dbSheet = ss.getSheetByName('База_JSON') || ss.insertSheet('База_JSON');
  dbSheet.clear();
  dbSheet.getRange('A1').setValue('JSON состояние (не удалять)');
  dbSheet.getRange('A2').setValue(body);
  dbSheet.getRange('B1').setValue('Обновлено:');
  dbSheet.getRange('B2').setValue(new Date());

  // 2. Формируем красивую таблицу "Рейтинг_Конкурса" для просмотра в Google Sheets
  var leaderSheet = ss.getSheetByName('Рейтинг_Конкурса') || ss.insertSheet('Рейтинг_Конкурса');
  leaderSheet.clear();
  var headers = ['Сотрудник', 'Часы за период', 'Сим', 'Абоны', 'Товарная выручка', 'Аксессуары', 'Фин услуги', 'Мега семья', 'Адаптер'];
  leaderSheet.appendRow(headers);
  leaderSheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#1e293b').setFontColor('#ffffff');

  if (data.contest && data.contest.participants) {
    data.contest.participants.forEach(function(p) {
      var f = p.facts || {};
      leaderSheet.appendRow([
        p.name,
        p.hours || 0,
        f.sim || 0,
        f.abon || 0,
        f.tovarka || 0,
        f.aks || 0,
        f.fin || 0,
        f.megasemya || 0,
        f.adapter || 0
      ]);
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', updated: new Date() }))
    .setMimeType(ContentService.MimeType.JSON);
}
